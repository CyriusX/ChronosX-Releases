# macOS Desktop Agent Porting Plan

## Executive Summary

This document outlines the comprehensive technical roadmap for porting the Windows-only TimeTrack Desktop Agent to macOS while maintaining full functional and visual parity with the existing Windows client. The solution will integrate seamlessly with the existing backend infrastructure without requiring server-side changes.

### 2.2 Technology Stack Decision

#### UI Framework: **SwiftUI + WKWebView** (Recommended)

**Rationale:**
- Native macOS look and feel
- Best performance and integration
- Full access to macOS APIs
- App Store compatible
- Smaller binary size vs MAUI

**Alternative: .NET MAUI**
- Single codebase for UI
- Less native feel
- Larger binary
- Still requires platform-specific code for system integration

**Recommendation:** Use SwiftUI for the native shell (menu bar, notifications, window management) with WKWebView hosting the existing React UI. This maximizes code reuse while providing native platform integration.

**File:** `TimeTrack.Agent.Infrastructure.MacOS/Providers/MacOSActiveWindowProvider.cs`

```csharp
using CoreFoundation;
using CoreGraphics;
using Foundation;
using System.Diagnostics;
using System.Runtime.InteropServices;
using System.Security.Cryptography;
using System.Text;

namespace TimeTrack.Agent.Infrastructure.Providers.MacOS;

public sealed class MacOSActiveWindowProvider : IActiveWindowProvider, IDisposable
{
    private readonly ILogger<MacOSActiveWindowProvider> _logger;
    private readonly IFilePathExtractor _filePathExtractor;
    private NSWorkspace _workspace;
    private IntPtr _notificationObserver;
    
    private ActiveWindowInfo? _cachedWindow;
    private DateTime _lastCacheUpdate = DateTime.MinValue;
    private readonly object _cacheLock = new();
    
    public event EventHandler<ActiveWindowChangedEventArgs>? ActiveWindowChanged;
    
    public MacOSActiveWindowProvider(
        ILogger<MacOSActiveWindowProvider> logger,
        IFilePathExtractor filePathExtractor)
    {
        _logger = logger;
        _filePathExtractor = filePathExtractor;
        _workspace = NSWorkspace.SharedWorkspace;
        
        // Subscribe to workspace notifications
        SetupWorkspaceNotifications();
    }
    
    private void SetupWorkspaceNotifications()
    {
        // NSWorkspaceDidActivateApplicationNotification
        var notificationCenter = _workspace.NotificationCenter;
        notificationCenter.AddObserver(
            NSWorkspace.DidActivateApplicationNotification,
            OnApplicationActivated,
            _workspace);
    }
    
    private void OnApplicationActivated(NSNotification notification)
    {
        var app = notification.UserInfo?[NSWorkspace.LaunchConfigurationApplicationName] as NSRunningApplication;
        if (app != null)
        {
            var info = BuildActiveWindowInfo(app);
            if (info != null)
            {
                UpdateCache(info);
                RaiseWindowChangedEvent(info);
            }
        }
    }
    
    public Task<ActiveWindowInfo?> GetActiveWindowAsync(CancellationToken cancellationToken = default)
    {
        if (IsCacheStale())
        {
            var info = PollActiveWindow();
            return Task.FromResult(info);
        }
        
        lock (_cacheLock)
        {
            return Task.FromResult(_cachedWindow);
        }
    }
    
    private ActiveWindowInfo? PollActiveWindow()
    {
        var frontmostApp = _workspace.FrontmostApplication;
        if (frontmostApp == null)
            return null;
            
        return BuildActiveWindowInfo(frontmostApp);
    }
    
    private ActiveWindowInfo? BuildActiveWindowInfo(NSRunningApplication app)
    {
        try
        {
            var exePath = app.BundleUrl?.Path ?? app.ExecutableUrl?.Path;
            if (string.IsNullOrEmpty(exePath))
                return null;
            
            var displayName = app.LocalizedName ?? Path.GetFileNameWithoutExtension(exePath);
            var exePathHash = ComputeSha256Hash(exePath);
            
            // Get window title using AXUIElement
            var windowTitle = GetActiveWindowTitle(app.ProcessIdentifier);
            var windowHash = !string.IsNullOrEmpty(windowTitle) 
                ? ComputeSha256Hash(windowTitle) 
                : null;
            
            // Extract file path
            var filePath = _filePathExtractor.ExtractFilePath(
                IntPtr.Zero, 
                displayName, 
                windowTitle);
            
            // Extract browser URL if applicable
            string? browserUrl = null;
            if (IsBrowserBundle(app.BundleIdentifier))
            {
                browserUrl = ExtractBrowserUrl(app);
            }
            
            return new ActiveWindowInfo
            {
                ExePathHash = exePathHash,
                DisplayName = displayName,
                ExePath = exePath,
                WindowTitle = windowTitle,
                WindowHash = windowHash,
                FilePath = filePath,
                BrowserUrl = browserUrl
            };
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error building ActiveWindowInfo for {AppName}", app.LocalizedName);
            return null;
        }
    }
    
    private string? GetActiveWindowTitle(int processId)
    {
        // Use AXUIElement API to get window title
        var appRef = AXUIElementCreateApplication(processId);
        if (appRef == IntPtr.Zero)
            return null;
        
        try
        {
            // Get focused window
            var focusedWindowRef = IntPtr.Zero;
            var result = AXUIElementCopyAttributeValue(
                appRef, 
                kAXFocusedWindowAttribute, 
                out focusedWindowRef);
            
            if (result != 0 || focusedWindowRef == IntPtr.Zero)
                return null;
            
            // Get title
            var titleValue = IntPtr.Zero;
            result = AXUIElementCopyAttributeValue(
                focusedWindowRef, 
                kAXTitleAttribute, 
                out titleValue);
            
            if (result != 0 || titleValue == IntPtr.Zero)
            {
                CFRelease(focusedWindowRef);
                return null;
            }
            
            var title = CFStringToString(titleValue);
            
            CFRelease(titleValue);
            CFRelease(focusedWindowRef);
            
            return title;
        }
        finally
        {
            CFRelease(appRef);
        }
    }
    
    // P/Invoke declarations for Accessibility APIs
    [DllImport("/System/Library/Frameworks/CoreFoundation.framework/CoreFoundation")]
    private static extern IntPtr AXUIElementCreateApplication(int pid);
    
    [DllImport("/System/Library/Frameworks/CoreFoundation.framework/CoreFoundation")]
    private static extern int AXUIElementCopyAttributeValue(IntPtr element, IntPtr attribute, out IntPtr value);
    
    [DllImport("/System/Library/Frameworks/CoreFoundation.framework/CoreFoundation")]
    private static extern void CFRelease(IntPtr cf);
    
    private static readonly IntPtr kAXFocusedWindowAttribute = CFStringCreate("AXFocusedWindow");
    private static readonly IntPtr kAXTitleAttribute = CFStringCreate("AXTitle");
    
    // ... helper methods
}
```

```csharp
using System.Runtime.InteropServices;

namespace TimeTrack.Agent.Infrastructure.Providers.MacOS;

public sealed class MacOSIdleDetector : IIdleDetector
{
    private readonly ILogger<MacOSIdleDetector> _logger;
    
    public event EventHandler<IdleStateChangedEventArgs>? IdleStateChanged;
    
    public MacOSIdleDetector(ILogger<MacOSIdleDetector> logger)
    {
        _logger = logger;
    }
    
    public Task<TimeSpan?> GetIdleTimeAsync(CancellationToken cancellationToken = default)
    {
        try
        {
            // Use CGEventSourceSecondsSinceLastEventType
            var idleSeconds = CGEventSourceSecondsSinceLastEventType(
                CGEventSourceStateID.CombinedSessionState,
                CGEventType.AllEvents);
            
            return Task.FromResult<TimeSpan?>(TimeSpan.FromSeconds(idleSeconds));
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error getting idle time");
            return Task.FromResult<TimeSpan?>(null);
        }
    }
    
    public async Task<bool> IsIdleAsync(TimeSpan threshold, CancellationToken cancellationToken = default)
    {
        var idleTime = await GetIdleTimeAsync(cancellationToken);
        return idleTime.HasValue && idleTime.Value >= threshold;
    }
    
    // CoreGraphics P/Invoke
    [DllImport("/System/Library/Frameworks/CoreGraphics.framework/CoreGraphics")]
    private static extern double CGEventSourceSecondsSinceLastEventType(
        int state, 
        int eventType);
    
    private const int CGEventSourceStateID_CombinedSessionState = 0;
    private const int CGEventType_AllEvents = -1;
}
```

```csharp
using System.Runtime.InteropServices;
using System.Text.Json;

namespace TimeTrack.Agent.Infrastructure.MacOS.Security;

public sealed class KeychainTokenStore : ITokenStore
{
    private const string ServiceName = "com.cyriusx.timetrack";
    private const string TokenAccount = "jwt_tokens";
    
    private readonly ILogger<KeychainTokenStore> _logger;
    private readonly HttpClient _httpClient;
    private readonly string _backendUrl;
    
    private TokenData? _cachedTokens;
    private readonly object _cacheLock = new();
    
    private sealed record TokenData(string Jwt, string RefreshToken, DateTime ExpiresAt);
    
    public event EventHandler<TokensStoredEventArgs>? TokensStored;
    public event EventHandler? TokensCleared;
    
    public KeychainTokenStore(
        ILogger<KeychainTokenStore> logger,
        HttpClient httpClient,
        SyncSettings settings)
    {
        _logger = logger;
        _httpClient = httpClient;
        _backendUrl = settings.BackendUrl;
    }
    
    public async Task<string?> GetJwtAsync(CancellationToken cancellationToken = default)
    {
        var tokens = await LoadTokensAsync(cancellationToken);
        return tokens?.Jwt;
    }
    
    public async Task<string?> GetRefreshTokenAsync(CancellationToken cancellationToken = default)
    {
        var tokens = await LoadTokensAsync(cancellationToken);
        return tokens?.RefreshToken;
    }
    
    public async Task StoreTokensAsync(string jwt, string refreshToken, CancellationToken cancellationToken = default)
    {
        var expiresAt = ExtractExpirationFromJwt(jwt);
        var tokens = new TokenData(jwt, refreshToken, expiresAt);
        
        var json = JsonSerializer.Serialize(tokens);
        var status = SecItemAddOrUpdate(ServiceName, TokenAccount, json);
        
        if (status != 0)
        {
            _logger.LogError("Failed to store tokens in Keychain: {Status}", status);
            throw new SecurityException("Failed to store tokens securely");
        }
        
        lock (_cacheLock)
        {
            _cachedTokens = tokens;
        }
        
        _logger.LogInformation("Tokens stored securely in macOS Keychain");
        
        TokensStored?.Invoke(this, new TokensStoredEventArgs
        {
            Jwt = jwt,
            RefreshToken = refreshToken
        });
    }
    
    public async Task ClearAsync(CancellationToken cancellationToken = default)
    {
        var status = SecItemDelete(ServiceName, TokenAccount);
        
        lock (_cacheLock)
        {
            _cachedTokens = null;
        }
        
        _logger.LogInformation("Tokens cleared from Keychain");
        TokensCleared?.Invoke(this, EventArgs.Empty);
    }
    
    public bool IsJwtExpiringSoon(int withinMinutes = 5)
    {
        lock (_cacheLock)
        {
            if (_cachedTokens is null)
                return true;
            return _cachedTokens.ExpiresAt <= DateTime.UtcNow.AddMinutes(withinMinutes);
        }
    }
    
    private async Task<TokenData?> LoadTokensAsync(CancellationToken cancellationToken)
    {
        lock (_cacheLock)
        {
            if (_cachedTokens is not null)
                return _cachedTokens;
        }
        
        var json = SecItemCopyContent(ServiceName, TokenAccount);
        if (json == null)
            return null;
        
        try
        {
            var tokens = JsonSerializer.Deserialize<TokenData>(json);
            lock (_cacheLock)
            {
                _cachedTokens = tokens;
            }
            return tokens;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to deserialize tokens from Keychain");
            return null;
        }
    }
    
    // Security framework P/Invoke
    [DllImport("/System/Library/Frameworks/Security.framework/Security")]
    private static extern int SecItemAdd(IntPtr attributes, IntPtr result);
    
    [DllImport("/System/Library/Frameworks/Security.framework/Security")]
    private static extern int SecItemDelete(IntPtr query);
    
    [DllImport("/System/Library/Frameworks/Security.framework/Security")]
    private static extern int SecItemCopyMatching(IntPtr query, out IntPtr result);
    
    // Helper methods for Keychain operations using CFDictionary
    // ...
}
```

```csharp
using System.Net.Sockets;
using System.Text.Json;

namespace TimeTrack.MacOSAgentService.Ipc;

public sealed class UnixDomainSocketIpcServer : IIpcServer, IHostedService, IDisposable
{
    private const string SocketPath = "/var/tmp/TimeTrack.Agent.IPC";
    
    private readonly ILogger<UnixDomainSocketIpcServer> _logger;
    private readonly IpcMessageRouter _messageRouter;
    private readonly Socket _serverSocket;
    private readonly CancellationTokenSource _cts = new();
    
    private Socket? _clientSocket;
    private StreamReader? _reader;
    private StreamWriter? _writer;
    private readonly object _writeLock = new();
    
    public bool IsListening { get; private set; }
    public bool IsClientConnected => _clientSocket?.Connected ?? false;
    
    public UnixDomainSocketIpcServer(
        ILogger<UnixDomainSocketIpcServer> logger,
        IpcMessageRouter messageRouter)
    {
        _logger = logger;
        _messageRouter = messageRouter;
        _serverSocket = new Socket(AddressFamily.Unix, SocketType.Stream, ProtocolType.IP);
    }
    
    public async Task StartAsync(CancellationToken cancellationToken)
    {
        // Remove existing socket file
        if (File.Exists(SocketPath))
            File.Delete(SocketPath);
        
        var endpoint = new UnixDomainSocketEndPoint(SocketPath);
        _serverSocket.Bind(endpoint);
        _serverSocket.Listen(1);
        
        // Set permissions: rw for owner and group
        File.SetUnixFileMode(SocketPath, 
            UnixFileMode.UserRead | UnixFileMode.UserWrite |
            UnixFileMode.GroupRead | UnixFileMode.GroupWrite);
        
        IsListening = true;
        _logger.LogInformation("IPC Server listening on {SocketPath}", SocketPath);
        
        // Start accepting connections
        _ = AcceptLoopAsync(_cts.Token);
    }
    
    private async Task AcceptLoopAsync(CancellationToken cancellationToken)
    {
        while (!cancellationToken.IsCancellationRequested)
        {
            try
            {
                var clientSocket = await _serverSocket.AcceptAsync(cancellationToken);
                _logger.LogInformation("Client connected");
                
                // Handle previous client
                if (_clientSocket != null)
                {
                    _reader?.Dispose();
                    _writer?.Dispose();
                    _clientSocket.Dispose();
                }
                
                _clientSocket = clientSocket;
                var stream = new NetworkStream(clientSocket);
                _reader = new StreamReader(stream, Encoding.UTF8);
                _writer = new StreamWriter(stream, Encoding.UTF8) { AutoFlush = true };
                
                // Start message processing loop
                _ = ProcessMessagesAsync(cancellationToken);
            }
            catch (OperationCanceledException)
            {
                break;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error accepting connection");
                await Task.Delay(1000, cancellationToken);
            }
        }
    }
    
    private async Task ProcessMessagesAsync(CancellationToken cancellationToken)
    {
        while (!cancellationToken.IsCancellationRequested && _clientSocket?.Connected == true)
        {
            try
            {
                var line = await _reader!.ReadLineAsync(cancellationToken);
                if (string.IsNullOrEmpty(line))
                {
                    // Client disconnected
                    _logger.LogInformation("Client disconnected");
                    break;
                }
                
                var request = JsonSerializer.Deserialize<IpcRequest>(line);
                if (request == null)
                {
                    _logger.LogWarning("Failed to deserialize IPC request");
                    continue;
                }
                
                var response = await _messageRouter.RouteAsync(request, cancellationToken);
                var responseJson = JsonSerializer.Serialize(response);
                
                lock (_writeLock)
                {
                    _writer!.WriteLine(responseJson);
                }
            }
            catch (IOException)
            {
                // Client disconnected
                break;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error processing message");
            }
        }
    }
    
    public async Task SendEventAsync(IpcEvent @event, CancellationToken cancellationToken = default)
    {
        if (_clientSocket?.Connected != true || _writer == null)
            return;
        
        var eventJson = JsonSerializer.Serialize(new { _event = @event });
        
        lock (_writeLock)
        {
            _writer.WriteLine(eventJson);
        }
    }
    
    public async Task StopAsync(CancellationToken cancellationToken)
    {
        _cts.Cancel();
        _reader?.Dispose();
        _writer?.Dispose();
        _clientSocket?.Dispose();
        _serverSocket.Dispose();
        
        if (File.Exists(SocketPath))
            File.Delete(SocketPath);
    }
```

#### 3.2.2 Unix Domain Socket IPC Client

**File:** `TimeTrack.MacOSDesktopHost/Ipc/UnixDomainSocketIpcClient.cs`

```csharp
using System.Net.Sockets;
using System.Text.Json;
using System.Collections.Concurrent;

namespace TimeTrack.MacOSDesktopHost.Ipc;

public sealed class UnixDomainSocketIpcClient : IIpcClient, IHostedService, IDisposable
{
    private const string SocketPath = "/var/tmp/TimeTrack.Agent.IPC";
    private const int ConnectionTimeoutMs = 5000;
    private const int ReconnectionDelayMs = 1000;
    private const int MaxReconnectionAttempts = 10;
    
    private readonly ILogger<UnixDomainSocketIpcClient> _logger;
    private readonly Socket _socket;
    private readonly CancellationTokenSource _cts = new();
    
    private StreamReader? _reader;
    private StreamWriter? _writer;
    private readonly object _writeLock = new();
    
    private int _nextRequestId = 1;
    private readonly ConcurrentDictionary<int, TaskCompletionSource<IpcResponse>> _pendingRequests = new();
    
    public bool IsConnected => _socket?.Connected ?? false;
    
    public event EventHandler<IpcEventArgs>? EventReceived;
    public event EventHandler<bool>? ConnectionStateChanged;
    
    public UnixDomainSocketIpcClient(ILogger<UnixDomainSocketIpcClient> logger)
    {
        _logger = logger;
        _socket = new Socket(AddressFamily.Unix, SocketType.Stream, ProtocolType.IP);
    }
    
    public async Task StartAsync(CancellationToken cancellationToken)
    {
        await ConnectAsync(cancellationToken);
        _ = ListenForEventsAsync(_cts.Token);
    }
    
    public async Task ConnectAsync(CancellationToken cancellationToken = default)
    {
        var attempts = 0;
        while (attempts < MaxReconnectionAttempts)
        {
            try
            {
                var endpoint = new UnixDomainSocketEndPoint(SocketPath);
                await _socket.ConnectAsync(endpoint, cancellationToken);
                
                var stream = new NetworkStream(_socket);
                _reader = new StreamReader(stream, Encoding.UTF8);
                _writer = new StreamWriter(stream, Encoding.UTF8) { AutoFlush = true };
                
                ConnectionStateChanged?.Invoke(this, true);
                _logger.LogInformation("Connected to AgentService");
                return;
            }
            catch (Exception ex)
            {
                attempts++;
                _logger.LogWarning(ex, "Connection attempt {Attempt} failed", attempts);
                await Task.Delay(ReconnectionDelayMs, cancellationToken);
            }
        }
        
        ConnectionStateChanged?.Invoke(this, false);
        throw new InvalidOperationException("Failed to connect to AgentService");
    }
    
    private async Task ListenForEventsAsync(CancellationToken cancellationToken)
    {
        while (!cancellationToken.IsCancellationRequested && _socket.Connected)
        {
            try
            {
                var line = await _reader!.ReadLineAsync(cancellationToken);
                if (string.IsNullOrEmpty(line))
                    continue;
                
                using var doc = JsonDocument.Parse(line);
                var root = doc.RootElement;
                
                if (root.TryGetProperty("_event", out var eventProp))
                {
                    // This is an event from the server
                    var eventType = eventProp.GetProperty("eventType").GetString()!;
                    var payload = eventProp.GetProperty("payload");
                    
                    EventReceived?.Invoke(this, new IpcEventArgs
                    {
                        EventType = eventType,
                        Payload = payload
                    });
                }
                else if (root.TryGetProperty("requestId", out var reqIdProp))
                {
                    // This is a response to a request
                    var requestId = reqIdProp.GetInt32();
                    var response = JsonSerializer.Deserialize<IpcResponse>(line);
                    
                    if (_pendingRequests.TryRemove(requestId, out var tcs))
                    {
                        tcs.SetResult(response!);
                    }
                }
            }
            catch (OperationCanceledException)
            {
                break;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error in event listener");
                await ReconnectAsync(cancellationToken);
            }
        }
    }
    
    public async Task<IpcResponse> SendCommandAsync(string command, object? payload = null, CancellationToken cancellationToken = default)
    {
        return await SendRequestAsync("command", command, payload, cancellationToken);
    }
    
    public async Task<IpcResponse> SendQueryAsync(string query, object? payload = null, CancellationToken cancellationToken = default)
    {
        return await SendRequestAsync("query", query, payload, cancellationToken);
    }
    
    private async Task<IpcResponse> SendRequestAsync(string type, string name, object? payload, CancellationToken cancellationToken)
    {
        var requestId = Interlocked.Increment(ref _nextRequestId);
        var tcs = new TaskCompletionSource<IpcResponse>();
        _pendingRequests[requestId] = tcs;
        
        var request = new
        {
            requestId,
            type,
            name,
            payload
        };
        
        var json = JsonSerializer.Serialize(request);
        
        lock (_writeLock)
        {
            _writer!.WriteLine(json);
        }
        
        // Timeout after 30 seconds
        using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        cts.CancelAfter(TimeSpan.FromSeconds(30));
        
        try
        {
            return await tcs.Task.WaitAsync(cts.Token);
        }
        catch (OperationCanceledException)
        {
            _pendingRequests.TryRemove(requestId, out _);
            throw;
        }
    }
    
    private async Task ReconnectAsync(CancellationToken cancellationToken)
    {
        ConnectionStateChanged?.Invoke(this, false);
        
        _reader?.Dispose();
        _writer?.Dispose();
        _socket.Disconnect(false);
        
        await ConnectAsync(cancellationToken);
    }
    
    public async Task StopAsync(CancellationToken cancellationToken)
    {
        _cts.Cancel();
        _reader?.Dispose();
        _writer?.Dispose();
        _socket.Dispose();
    }
    
    public void Dispose()
    {
        _cts.Dispose();
        _socket.Dispose();
    }
}
```

```swift
struct ContentView: View {

struct WebViewContainer: NSViewRepresentable {
    @ObservedObject var ipcClient: IpcClient
    @Binding var webView: WKWebView?
    
    func makeNSView(context: Context) -> WKWebView {
        let config = WKWebViewConfiguration()
        let userContentController = WKUserContentController()
        
        // Inject bridge script BEFORE page loads
        let bridgeScript = WKUserScript(
            source: Self.bridgeJavaScript,
            injectionTime: .atDocumentStart,
            forMainFrameOnly: true
        )
        userContentController.addUserScript(bridgeScript)
        
        // Register message handler for IPC
        userContentController.add(context.coordinator, name: "timeTrackBridge")
        
        config.userContentController = userContentController
        config.preferences.setValue(true, forKey: "developerExtrasEnabled")
        
        let webView = WKWebView(frame: .zero, configuration: config)
        self.webView = webView
        
        // Load the UI bundle
        if let bundlePath = Bundle.main.path(forResource: "dist", ofType: nil) {
            let url = URL(fileURLWithPath: bundlePath).appendingPathComponent("index.html")
            webView.loadFileURL(url, allowingReadAccessTo: URL(fileURLWithPath: bundlePath))
        }
        
        return webView
    }
    
    func updateNSView(_ nsView: WKWebView, context: Context) {}
    
    func makeCoordinator() -> Coordinator {
        Coordinator(ipcClient: ipcClient)
    }
    
    class Coordinator: NSObject, WKScriptMessageHandler {
        let ipcClient: IpcClient
        
        init(ipcClient: IpcClient) {
            self.ipcClient = ipcClient
        }
        
        func userContentController(_ userContentController: WKUserContentController, didReceive message: WKScriptMessage) {
            guard let body = message.body as? [String: Any],
                  let type = body["type"] as? String,
                  let name = body["name"] as? String,
                  let requestId = body["requestId"] as? Int else {
                return
            }
            
            Task {
                do {
                    var response: IpcResponse
                    if type == "command" {
                        response = try await ipcClient.sendCommand(name, payload: body["payload"])
                    } else {
                        response = try await ipcClient.sendQuery(name, payload: body["payload"])
                    }
                    
                    let responseJson = String(data: try JSONEncoder().encode(response), encoding: .utf8) ?? "{}"
                    let script = "window._timeTrackIpcResponse(\(requestId), \(responseJson))"
                    
                    await message.webView?.evaluateJavaScript(script)
                } catch {
                    let errorResponse = "{\"success\":false,\"error\":\"\(error.localizedDescription)\"}"
                    let script = "window._timeTrackIpcResponse(\(requestId), \(errorResponse))"
                    await message.webView?.evaluateJavaScript(script)
                }
            }
        }
    }
    
    static let bridgeJavaScript = """
    (function() {
        var _nextId = 1;
        var _pending = {};
        
        window._timeTrackIpcResponse = function(requestId, responseJson) {
            var p = _pending[requestId];
            if (p) {
                delete _pending[requestId];
                p(JSON.parse(responseJson));
            }
        };
        
        function ipcCall(type, name, payload) {
            return new Promise(function(resolve) {
                var id = _nextId++;
                _pending[id] = resolve;
                window.webkit.messageHandlers.timeTrackBridge.postMessage({
                    requestId: id,
                    type: type,
                    name: name,
                    payload: payload || null
                });
            });
        }
        
        window.timeTrackBridge = {
            isConnected: true,
            SendCommand: function(command, payload) { return ipcCall('command', command, payload); },
            SendQuery: function(query, payload) { return ipcCall('query', query, payload); }
        };
        
        console.log('[MacOSDesktopHost] Bridge injected successfully');
    })();
    """
}
```

## 5. Risk Assessment & Mitigation

| Risk | Impact | Mitigation |
|------|--------|------------|
| Accessibility permission denied | High - Core tracking fails | Graceful degradation, clear user guidance, fallback to polling |
| Sandbox restrictions | Medium - May limit IPC | Use NSTemporaryDirectory() for socket, request appropriate entitlements |
| WKWebView limitations | Low - Bridge patterns well-established | Test edge cases, maintain WebView2 bridge API compatibility |
| Sparkle update failures | Medium - Users stuck on old version | Fallback to manual download, robust error handling |
| Notarization rejection | High - Can't distribute | Follow Apple guidelines, test early, use hardened runtime |
